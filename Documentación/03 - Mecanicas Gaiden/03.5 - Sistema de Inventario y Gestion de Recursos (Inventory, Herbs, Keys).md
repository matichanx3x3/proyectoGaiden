# 03.5 - Sistema de Inventario y Gestión de Recursos (Inventory & Consumables)

Fiel a las raíces del *Survival Horror*, *Resident Evil Gaiden* impone una severa limitación de recursos:
- Un maletín con un máximo de **8 casillas para objetos** (`items[8]` en la ROM).
- Munición agrupada por tipo de arma con tope en 99 unidades.
- Sistema de hierbas curativas tradicionales (Verde, Roja, Mezclada) y Spray de Primeros Auxilios.
- Llaves maestras y tarjetas de acceso necesarias para desbloquear puertas cerradas del trasatlántico.

---

## 🎒 Catálogo de Objetos de Gaiden

| Objeto | Tipo | Efecto en Juego |
| :--- | :--- | :--- |
| **Hierba Verde** (`GreenHerb`) | Consumible | Restaura **30 HP** al personaje activo. |
| **Hierba Roja** (`RedHerb`) | Material | No usable sola; se combina con Hierba Verde para curación máxima. |
| **Hierba Mixta** (`MixedHerb`) | Consumible | Restaura **100% HP** y cura veneno. |
| **Spray Primeros Auxilios** (`FirstAidSpray`) | Consumible | Restaura **100 HP** y elimina el estado de veneno (`is_poisoned = false`). |
| **Chaleco Antibalas** (`BodyArmor`) | Pasivo | Reduce un 25% el daño recibido por zombies. |
| **Llave del Capitán** (`CaptainsKey`) | Llave | Abre el puente de mando y camarotes VIP. |
| **Llave de Pasajeros** (`PassengerKey`) | Llave | Desbloquea habitaciones de la Cubierta B. |
| **Llave de Calderas** (`BoilerKey`) | Llave | Abre la sala de máquinas inferior. |
| **Tarjeta de Seguridad** (`SecurityCard`) | Llave | Permite el acceso a los laboratorios ocultos de Umbrella. |

## 💻 Implementación en Unity C# (`InventorySystem.cs`)

```csharp
using System;
using System.Collections.Generic;
using UnityEngine;
using Game.Core;
using Game.Weapons;
using Game.Units;

namespace Game.Items {
    public enum ItemId {
        None = 0,
        GreenHerb = 1,
        RedHerb = 2,
        MixedHerb = 3,
        FirstAidMed = 4,
        FirstAidSpray = 5,
        BodyArmor = 6,
        CaptainsKey = 7,
        PassengerKey = 8,
        BoilerKey = 9,
        KeyCard_Level1 = 10,
        KeyCard_Level2 = 11
    }

    public class InventorySystem : Singleton<InventorySystem> {
        public static event Action OnInventoryUpdated;

        private const int MAX_SLOTS = 8;
        [SerializeField] private List<ItemId> _slots = new List<ItemId>(MAX_SLOTS);
        private Dictionary<WeaponType, int> _ammo = new Dictionary<WeaponType, int>();

        public ScriptableWeapon EquippedWeapon { get; private set; }
        public IReadOnlyList<ItemId> Slots => _slots;

        protected override void Awake() {
            base.Awake();
            InitializeStartingInventory();
        }

        private void InitializeStartingInventory() {
            _ammo[WeaponType.MeleeKnife] = 999;
            _ammo[WeaponType.Handgun] = 30;
            _ammo[WeaponType.Shotgun] = 8;
            _ammo[WeaponType.Rifle] = 0;
            _ammo[WeaponType.HeavyExplosive] = 0;

            if (ResourceSystem.Instance != null) {
                EquippedWeapon = ResourceSystem.Instance.GetWeapon(WeaponType.Handgun);
            }

            if (_slots.Count == 0) {
                _slots.Add(ItemId.FirstAidMed);
                _slots.Add(ItemId.GreenHerb);
            }
        }

        public bool AddItem(ItemId id) {
            if (_slots.Count >= MAX_SLOTS) {
                Debug.LogWarning("[Inventory] Maletín lleno. No se puede recoger el objeto.");
                return false;
            }
            _slots.Add(id);
            OnInventoryUpdated?.Invoke();
            return true;
        }

        public bool HasItem(ItemId id) => _slots.Contains(id);

        public bool RemoveItem(ItemId id) {
            bool removed = _slots.Remove(id);
            if (removed) OnInventoryUpdated?.Invoke();
            return removed;
        }

        public void UseItem(int slotIndex) {
            if (slotIndex < 0 || slotIndex >= _slots.Count) return;

            ItemId item = _slots[slotIndex];
            var activeHero = PartyManager.Instance != null ? PartyManager.Instance.ActiveHero : null;
            if (activeHero == null) return;

            switch (item) {
                case ItemId.FirstAidMed:
                case ItemId.FirstAidSpray:
                    activeHero.Heal(100);
                    PartyManager.Instance.CurePoison();
                    _slots.RemoveAt(slotIndex);
                    break;
                case ItemId.GreenHerb:
                    activeHero.Heal(30);
                    _slots.RemoveAt(slotIndex);
                    break;
                case ItemId.MixedHerb:
                    activeHero.Heal(120);
                    PartyManager.Instance.CurePoison();
                    _slots.RemoveAt(slotIndex);
                    break;
                default:
                    Debug.Log($"[Inventory] {item} no se puede usar directamente desde aquí.");
                    return;
            }

            OnInventoryUpdated?.Invoke();
        }

        public bool DropItem(int index, Vector3 dropPosition) {
            if (index < 0 || index >= _slots.Count) return false;
            ItemId item = _slots[index];
            _slots.RemoveAt(index);

            var pickupGo = GameObject.CreatePrimitive(PrimitiveType.Cube);
            pickupGo.name = $"Pickup_{item}";
            pickupGo.transform.position = dropPosition + Vector3.up * 0.35f;
            pickupGo.transform.localScale = new Vector3(0.35f, 0.35f, 0.35f);

            var col = pickupGo.GetComponent<Collider>();
            if (col != null) col.isTrigger = true;

            var pickup = pickupGo.AddComponent<Game.Exploration.ItemPickup>();
            pickup.Configure(item);

            OnInventoryUpdated?.Invoke();
            Debug.Log($"[Inventory] Objeto soltado al escenario: {item} en {dropPosition}");
            return true;
        }

        public int GetAmmo(WeaponType type) => _ammo.TryGetValue(type, out int val) ? val : 0;

        public void AddAmmo(WeaponType type, int amount) {
            if (!_ammo.ContainsKey(type)) _ammo[type] = 0;
            _ammo[type] = Mathf.Clamp(_ammo[type] + amount, 0, 99);
            OnInventoryUpdated?.Invoke();
        }

        public void ConsumeAmmo(WeaponType type, int amount = 1) {
            if (type == WeaponType.MeleeKnife) return;
            if (_ammo.ContainsKey(type)) {
                _ammo[type] = Mathf.Max(0, _ammo[type] - amount);
                OnInventoryUpdated?.Invoke();
            }
        }
    }
}
```

## 🌿 Recogida de Objetos en el Suelo (`ItemPickup.cs`)

Para resolver el problema común de objetos que bloquean físicamente al personaje como paredes, `ItemPickup` configura un colisionador tridimensional con `isTrigger = true` y ofrece una animación matemática diegética de flotación:

```csharp
using UnityEngine;
using Game.Interfaces;
using Game.Items;
using Game.Exploration;

namespace Game.Exploration {
    [RequireComponent(typeof(Collider))]
    public class ItemPickup : MonoBehaviour, IInteractable {
        [Header("Configuración del Objeto")]
        [SerializeField] private ItemId _itemToGive = ItemId.GreenHerb;
        [SerializeField] private int _quantity = 1;

        [Header("Animación Flotante (Diegética)")]
        [SerializeField] private float _bobSpeed = 2.5f;
        [SerializeField] private float _bobHeight = 0.15f;
        [SerializeField] private float _rotateSpeed = 45f;

        private Vector3 _startPosition;
        private bool _isCollected;

        private void Start() {
            _startPosition = transform.position;
            var col = GetComponent<Collider>();
            if (col != null) col.isTrigger = true;
        }

        private void Update() {
            if (_isCollected) return;
            float newY = _startPosition.y + Mathf.Sin(Time.time * _bobSpeed) * _bobHeight;
            transform.position = new Vector3(_startPosition.x, newY, _startPosition.z);
            transform.Rotate(Vector3.up, _rotateSpeed * Time.deltaTime, Space.World);
        }

        // 1. Recogida automática al pasar por encima
        private void OnTriggerEnter(Collider other) {
            if (_isCollected) return;
            if (other.CompareTag("Player") || other.GetComponent<ExplorationController>() != null) {
                ExecutePickup();
            }
        }

        // 2. Interacción manual (Tecla E / Botón de Acción)
        public void Interact(ExplorationController player) {
            if (_isCollected) return;
            ExecutePickup();
        }

        private void ExecutePickup() {
            if (InventorySystem.Instance != null) {
                bool added = InventorySystem.Instance.AddItem(_itemToGive);
                if (added) {
                    _isCollected = true;
                    Debug.Log($"[ItemPickup] Objeto recogido con éxito: {_itemToGive}");
                    Destroy(gameObject);
                }
            }
        }

        public string GetInteractionPrompt() => $"Recoger {_itemToGive}";
    }
}
```

## 🖥️ Interfaz de Usuario y Debounce (`InventoryUI.cs`)

Para evitar bucles infinitos donde la misma pulsación abre y cierra el menú inmediatamente, `InventoryUI` implementa un *debounce* de `0.25s` utilizando `Time.unscaledTime`:

```csharp
using System.Collections.Generic;
using UnityEngine;
using UnityEngine.UI;
using UnityEngine.InputSystem;
using Game.Core;
using Game.Items;
using Game.Units;

namespace Game.UI {
    public class InventoryUI : MonoBehaviour {
        [SerializeField] private GameObject _inventoryPanel;
        [SerializeField] private List<Text> _slotTexts = new List<Text>();
        [SerializeField] private Text _itemDescriptionText;
        [SerializeField] private Text _partyStatusText;

        private float _openedTime;
        private int _selectedSlotIndex = 0;

        private void HandleStateChanged(GameState newState) {
            bool isInv = (newState == GameState.Inventory);
            if (_inventoryPanel != null) _inventoryPanel.SetActive(isInv);
            if (isInv) {
                _openedTime = Time.unscaledTime;
                RefreshSlots();
            }
        }

        private void Update() {
            if (_inventoryPanel == null || !_inventoryPanel.activeSelf) return;

            // Debounce anti-rebote: ignora inputs en los primeros 0.25s de apertura
            if (Time.unscaledTime - _openedTime < 0.25f) return;

            var keyboard = Keyboard.current;
            if (keyboard != null) {
                bool isShift = keyboard.leftShiftKey.isPressed || keyboard.rightShiftKey.isPressed;
                if (keyboard.digit1Key.wasPressedThisFrame) HandleSlotAction(0, isShift);
                // ... slots 2 a 8 ...

                // Atajo [D] o [X] para soltar objeto al escenario
                if (keyboard.dKey.wasPressedThisFrame || keyboard.xKey.wasPressedThisFrame) {
                    DropSlot(_selectedSlotIndex);
                }

                if (keyboard.iKey.wasPressedThisFrame || keyboard.tabKey.wasPressedThisFrame || keyboard.escapeKey.wasPressedThisFrame) {
                    CloseInventory();
                }
            }

            var gamepad = Gamepad.current;
            if (gamepad != null) {
                if (gamepad.buttonEast.wasPressedThisFrame || gamepad.startButton.wasPressedThisFrame) CloseInventory();
                if (gamepad.buttonNorth.wasPressedThisFrame) DropSlot(_selectedSlotIndex);
            }
        }

        public void DropSlot(int index) {
            var inv = InventorySystem.Instance;
            if (inv == null || index < 0 || index >= inv.Slots.Count) return;

            Vector3 dropPos = Vector3.zero;
            var player = GameObject.FindGameObjectWithTag("Player");
            if (player != null) dropPos = player.transform.position + player.transform.forward * 1.0f;

            inv.DropItem(index, dropPos);
            RefreshSlots();
        }

        public void CloseInventory() {
            if (GameManager.Instance != null) {
                GameManager.Instance.ChangeState(GameState.Exploration);
            }
        }
    }
}
```
