# 03.2 - Sistema de Combate y Retícula Oscilante (Combat Engine & Reticle)

El sistema de combate de *Resident Evil Gaiden* es el elemento más distintivo del juego. Al comenzar el combate, la perspectiva cambia a **primera persona**, mostrando al enemigo de frente y una **barra horizontal de sincronización** en la parte inferior de la pantalla.

---

## 🎯 Anatomía del Retículo de Combate

Basado en la ingeniería inversa de la ROM (`decompgaiden/src/combat.c` y Banco de ROM `0D` `0x409C` / `0x49C2`):

```
       [======================== GAUGE: 0 .. 120 px ========================]
       0                                60                                120
       |---------------------[   (  |  )   ]--------------------------------|
                             |   |  |  |   |
                             |   |  |  |   +-- Fin Zona de Impacto Normal (Hit Width)
                             |   |  |  +------ Fin Diana Crítica (Crit Width)
                             |   |  +--------- Centro del Objetivo (Target Center = 60)
                             |   +------------ Inicio Diana Crítica
                             +---------------- Inicio Zona Normal
                                    ^
                                  Aguja blanca oscilante (Speed: 3 px/frame)
```

## 🧮 Algoritmo Matemático de Impacto

Cuando el jugador aprieta el gatillo (Botón **A**):

$$dist = |\text{posicionAguja} - \text{centroObjetivo}|$$

$$\text{Resultado} = \begin{cases} 
\text{CRÍTICO} & \text{si } dist \le \text{CritHalfWidth} \\ 
\text{NORMAL} & \text{si } dist \le \text{HitHalfWidth} \\ 
\text{FALLO} & \text{en cualquier otro caso} 
\end{cases}$$

### Fórmula de Daño (Banco 0D: 0x49C2):
$$\text{Daño} = \begin{cases} 
0 & \text{si Fallo} \\ 
\text{DañoBase} & \text{si Impacto Normal} \\ 
\text{DañoBase} \times \text{CritMultiplier} & \text{si Crítico} 
\end{cases}$$

## 💻 Implementación en Unity C# (`CombatManager.cs`)

```csharp
using System;
using UnityEngine;
using Gaiden.Core;
using Gaiden.Weapons;
using Gaiden.Units;

namespace Gaiden.Combat {
    public enum HitOutcome { Miss, Normal, Critical }

    public class CombatManager : Singleton<CombatManager> {
        public static event Action<ScriptableWeapon, bool> OnShotFired;
        public static event Action<HitOutcome, int> OnEnemyHit;
        public static event Action<int> OnPlayerDamaged;
        public static event Action OnCombatEnded;

        [Header("Configuración de Retículo")]
        [SerializeField] private float _gaugeMaxWidth = 120f;
        [SerializeField] private float _reticleSpeed = 180f; // Píxeles por segundo

        // Estado activo
        public ScriptableEnemy CurrentEnemy { get; private set; }
        public int CurrentEnemyHp { get; private set; }
        public float NeedlePosition { get; private set; }
        public float TargetCenter { get; private set; } = 60f;
        public float TurnTimer { get; private set; }

        private int _needleDirection = 1;
        private bool _isCombatActive;
        private EnemyOverworldUnit _overworldReference;

        public void StartCombat(EnemyType type, EnemyOverworldUnit overworldRef) {
            CurrentEnemy = ResourceSystem.Instance.GetEnemy(type);
            CurrentEnemyHp = CurrentEnemy.MaxHealth;
            _overworldReference = overworldRef;

            NeedlePosition = 0f;
            _needleDirection = 1;
            TurnTimer = CurrentEnemy.AttackIntervalFrames / 60f; // Convertir frames a segundos
            _isCombatActive = true;

            GaidenGameManager.Instance.ChangeState(GaidenGameState.Combat);
        }

        private void Update() {
            if (!_isCombatActive) return;

            UpdateNeedle();
            UpdateEnemyTurn();
            HandleCombatInput();
        }

        private void UpdateNeedle() {
            NeedlePosition += _needleDirection * _reticleSpeed * Time.deltaTime;

            if (NeedlePosition >= _gaugeMaxWidth) {
                NeedlePosition = _gaugeMaxWidth;
                _needleDirection = -1;
            } else if (NeedlePosition <= 0f) {
                NeedlePosition = 0f;
                _needleDirection = 1;
            }
        }

        private void UpdateEnemyTurn() {
            TurnTimer -= Time.deltaTime;
            if (TurnTimer <= 0f) {
                // El zombie contraataca
                TurnTimer = CurrentEnemy.AttackIntervalFrames / 60f;
                PartyManager.Instance.ActiveHero.TakeDamage(CurrentEnemy.AttackPower);
                OnPlayerDamaged?.Invoke(CurrentEnemy.AttackPower);
            }
        }

        private void HandleCombatInput() {
            // Disparar arma (Botón A / Espacio)
            if (Input.GetButtonDown("Fire1") || Input.GetKeyDown(KeyCode.Space)) {
                PullTrigger();
            }

            // Cambiar de arma (Botón B / Q)
            if (Input.GetButtonDown("Fire2") || Input.GetKeyDown(KeyCode.Q)) {
                InventorySystem.Instance.CycleNextWeapon();
            }

            // Cambiar de personaje en combate (Select / Tab)
            if (Input.GetKeyDown(KeyCode.Tab)) {
                PartyManager.Instance.CycleNextHero();
            }
        }

        public void PullTrigger() {
            var weapon = InventorySystem.Instance.EquippedWeapon;

            // Verificar munición (el cuchillo tiene munición infinita)
            if (weapon.WeaponType != WeaponType.Knife && InventorySystem.Instance.GetAmmo(weapon.WeaponType) <= 0) {
                OnShotFired?.Invoke(weapon, false); // Click sin balas
                return;
            }

            // Descontar bala
            if (weapon.WeaponType != WeaponType.Knife) {
                InventorySystem.Instance.ConsumeAmmo(weapon.WeaponType, 1);
            }

            OnShotFired?.Invoke(weapon, true);

            // Calcular precisión
            float dist = Mathf.Abs(NeedlePosition - TargetCenter);
            HitOutcome outcome = HitOutcome.Miss;

            if (dist <= CurrentEnemy.CritHalfWidth) {
                outcome = HitOutcome.Critical;
            } else if (dist <= CurrentEnemy.HitHalfWidth) {
                outcome = HitOutcome.Normal;
            }

            int damage = 0;
            if (outcome == HitOutcome.Critical) {
                damage = weapon.BaseDamage * weapon.CritMultiplier;
            } else if (outcome == HitOutcome.Normal) {
                damage = weapon.BaseDamage;
            }

            if (damage > 0) {
                CurrentEnemyHp = Mathf.Max(0, CurrentEnemyHp - damage);
            }

            OnEnemyHit?.Invoke(outcome, damage);

            // Comprobar muerte del enemigo
            if (CurrentEnemyHp <= 0) {
                EndCombat(victory: true);
            }
        }

        private void EndCombat(bool victory) {
            _isCombatActive = false;
            if (victory && _overworldReference != null) {
                UnitManager.Instance.RemoveEnemy(_overworldReference);
            }

            OnCombatEnded?.Invoke();
            GaidenGameManager.Instance.ChangeState(GaidenGameState.Exploration);
        }
    }
}
```
